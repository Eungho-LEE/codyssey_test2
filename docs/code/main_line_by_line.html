def read_number(prompt, minimum, maximum):
    while True:
        raw = input(prompt).strip()
        if not raw:
            print("빈 입력입니다. 숫자를 입력하세요.")
            continue
        try:
            value = int(raw)
        except ValueError:
            print("정수를 입력하세요.")
            continue
        if minimum <= value <= maximum:
            return value
        print(f"{minimum}~{maximum} 범위로 입력하세요.")


def read_nonempty(prompt):

        if value:
            return value

class Quiz:
    def __init__(self, question, choices, answer):
        if not isinstance(question, str) or not question.strip():
            raise ValueError("문제는 빈 문자열일 수 없습니다.")
        if not isinstance(choices, list) or len(choices) != 4:
            raise ValueError("선택지는 4개여야 합니다.")
        if any(not isinstance(item, str) or not item.strip() for item in choices):
            raise ValueError("선택지는 빈 문자열일 수 없습니다.")
        if type(answer) is not int or not 1 <= answer <= 4:
            raise ValueError("정답은 1~4 정수여야 합니다.")
        self.question = question.strip()
        self.choices = [item.strip() for item in choices]
        self.answer = answer

    def show(self):
        print(self.question)
        for number, text in enumerate(self.choices, start=1):
            print(f"{number}. {text}")

    def is_correct(self, selected):
        return selected == self.answer

    def to_dict(self):
        return {"question": self.question,
                "choices": self.choices, "answer": self.answer}


    @classmethod
    def from_dict(cls, data):
        if not isinstance(data, dict):
            raise ValueError("퀴즈 항목은 객체여야 합니다.")
        if not all(key in data for key in ("question", "choices", "answer")):
            raise ValueError("퀴즈 필드가 누락됐습니다.")
        return cls(data["question"], data["choices"], data["answer"])


class QuizGame:

    def play(self):
        if not self.quizzes:
            print("등록된 퀴즈가 없습니다.")
            return
        correct = 0
        for quiz in self.quizzes:
            quiz.show()
            selected = read_number("정답 번호: ", 1, 4)
            if quiz.is_correct(selected):
                correct += 1
                print("정답입니다.")
            else:
                print("오답입니다.")
        print(f"{len(self.quizzes)}문제 중 {correct}개 정답")
        if self.best_score is None or correct > self.best_score:
            self.best_score = correct
            print("새로운 최고 기록입니다.")
        if self.save():
            print("점수를 저장했습니다.")
        else:
            print("점수 저장 실패: 메모리에만 남아 있습니다.")


    def add_quiz(self):

        quiz = Quiz(question, choices, answer)
        self.quizzes.append(quiz)
        print("퀴즈를 목록에 추가했습니다.")
        if self.save():
            print("추가한 퀴즈를 파일에도 저장했습니다.")
        else:
            print("파일 저장 실패: 추가한 퀴즈는 메모리에만 있습니다.")


    def list_quizzes(self):
        if not self.quizzes:
            print("등록된 퀴즈가 없습니다.")
            return
        print(f"등록된 퀴즈: {len(self.quizzes)}개")
        for number, quiz in enumerate(self.quizzes, start=1):
            print(f"{number}. {quiz.question}")


    def show_score(self):
        if self.best_score is None:
            print("아직 퀴즈를 풀지 않았습니다.")
        else:
            print(f"최고 정답 수: {self.best_score}개")


    def save(self):
        if not self.can_save:
            print("원본 보호를 위해 저장을 보류했습니다.")
            return False
        payload = {"quizzes": [q.to_dict() for q in self.quizzes],
                   "best_score": self.best_score}
        temporary = STATE_PATH.with_suffix(".json.tmp")
        try:
            with temporary.open("w", encoding="utf-8") as file:
                json.dump(payload, file, ensure_ascii=False, indent=2)
            temporary.replace(STATE_PATH)
        except OSError as error:
            print(f"저장하지 못했습니다: {error}")
            return False
        return True


    def parse_state(self, data):
        if not isinstance(data, dict):
            raise ValueError("전체 데이터는 객체여야 합니다.")
        if not isinstance(data.get("quizzes"), list) or "best_score" not in data:
            raise ValueError("quizzes 또는 best_score 구조가 잘못됐습니다.")
        quizzes = [Quiz.from_dict(item) for item in data["quizzes"]]
        score = data["best_score"]
        if score is not None and (type(score) is not int or score < 0):
            raise ValueError("최고 점수는 null 또는 0 이상 정수여야 합니다.")
        return quizzes, score


    def load(self):
        try:
            with STATE_PATH.open("r", encoding="utf-8") as file:
                data = json.load(file)
            quizzes, score = self.parse_state(data)
        except FileNotFoundError:
            print("첫 실행: 기본 퀴즈를 사용합니다.")
            return
        except (json.JSONDecodeError, UnicodeDecodeError, ValueError) as error:
            print(f"상태 파일 손상: {error}")
            backup = STATE_PATH.with_name(
                "state.corrupt-" + datetime.now().strftime("%Y%m%d-%H%M%S-%f") + ".json"
            )
            try:
                shutil.copy2(STATE_PATH, backup)
                print(f"손상 원본 백업: {backup.name}")
            except OSError as backup_error:
                self.can_save = False
                print(f"백업 실패: 원본을 덮어쓰지 않습니다. {backup_error}")
            print("이번 실행은 기본 퀴즈로 복구합니다.")
            return
        except OSError as error:
            self.can_save = False
            print(f"읽기 실패: 기본 퀴즈를 사용하고 저장은 보류합니다. {error}")
            return
        self.quizzes, self.best_score = quizzes, score
        print("저장된 데이터를 불러왔습니다.")

    def run(self):
        while True:
            print("\n1. 퀴즈 풀기 / 2. 추가 / 3. 목록 / 4. 점수 / 5. 종료")
            choice = read_number("선택: ", 1, 5)
            if choice == 5:
                print("종료합니다.")
                return
            elif choice == 1:
                self.play()
            elif choice == 2:
                self.add_quiz()
            elif choice == 3:
                self.list_quizzes()
            elif choice == 4:
                self.show_score()
            else:
                print("이 기능은 다음 단계에서 연결합니다.")


def main():
    game = QuizGame()

    finally:

            print("저장 실패 또는 보류: 이번 변경이 파일에 남지 않았을 수 있습니다.")


if __name__ == "__main__":

prompt는 안내 문구, minimum/maximum은 허용 범위입니다. 공백·빈 입력·숫자 변환·범위를 검사하고 정상 정수만 반환합니다.

read_nonempty

빈 문장이 아닌 입력 받기

문제와 선택지 입력에 재사용합니다. 공백만 입력하면 다시 받고, 정상 문자열을 return합니다.

Quiz

문제 하나의 데이터와 동작

question·choices·answer를 함께 보관하고 출력·판정·dict 변환을 담당합니다.

Quiz.__init__

문제 객체 생성과 검증

문자열 문제, 선택지 네 개, 정답 번호 1~4를 검증한 뒤 속성으로 저장합니다.

Quiz.show

문제와 선택지 출력

문제 문장과 1부터 시작하는 선택지 번호를 보여 줍니다.

Quiz.is_correct

정답 비교

선택한 정수 번호와 저장된 정답 번호가 같은지 bool로 반환합니다.

Quiz.to_dict

JSON에 저장할 기본 자료형으로 변환

사용자 정의 Quiz 객체를 dict/list/문자열/정수로 바꿉니다. 이 메서드 자체는 파일을 쓰지 않습니다.

Quiz.from_dict

저장된 문제 복원

필수 키를 검사한 dict에서 Quiz 객체를 새로 만듭니다. 생성자의 검증도 다시 거칩니다.

QuizGame

게임 전체 상태와 기능

문제 목록·최고 정답 수·저장 가능 상태를 관리하고 메뉴와 파일 입출력을 연결합니다.

QuizGame.__init__

게임 준비

기본 퀴즈·미플레이 점수·저장 허용 상태를 먼저 만들고 저장 파일을 불러옵니다.

QuizGame.make_default_quizzes

개발 기초 문제 다섯 개 준비

처음 실행하거나 손상 파일에서 복구할 때 사용할 새 Quiz 객체 목록을 반환합니다.

QuizGame.play

한 판 진행

문제를 차례로 출제하고 정답 수를 세며, 완료한 판에 대해 최고점을 갱신하고 저장합니다.

QuizGame.add_quiz

문제 등록

모든 필드를 정상 입력받은 뒤 객체 하나를 목록에 추가하고 파일에 저장합니다.

QuizGame.list_quizzes

문제 목록 확인

빈 목록을 처리하고, 등록된 문제 문장에 1부터 번호를 붙여 출력합니다.

QuizGame.show_score

최고 정답 수 확인

아직 플레이하지 않은 None과 실제 정답 수 0을 구분합니다.

QuizGame.save

상태 저장

저장이 허용될 때 dict로 변환하여 임시 파일을 완성한 뒤 state.json과 교체합니다. 성공 여부를 bool로 돌려줍니다.

QuizGame.parse_state

저장 구조 검증

전체 구조·문제 목록·최고점의 타입을 검사합니다. 게임 상태를 직접 바꾸지 않고 검증한 목록과 점수를 반환합니다.

QuizGame.load

상태 읽기와 오류 복구

정상 파일을 검증해 적용합니다. 파일 없음·손상·읽기 실패를 구별하고 손상 원본을 백업합니다.

QuizGame.run

메뉴 반복

검증된 메뉴 번호로 각 메서드를 호출하고, 5번이면 run만 끝내 main으로 돌아갑니다.

main

실행 시작과 종료 처리

게임을 준비하고 메뉴를 실행합니다. Ctrl+C·EOF를 안내하고 종료 시 가능한 저장을 다시 수행합니다.
